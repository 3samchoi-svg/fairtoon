insert into public.work_legal_links (work_id, platform_key, url, sort)
select w.id, 'ridi', imp.url, 1
  from imp
  join public.works w on w.title_ko = imp.title
 where not exists (
   select 1 from public.work_legal_links l
    where l.work_id = w.id and l.platform_key = 'ridi'
 );
