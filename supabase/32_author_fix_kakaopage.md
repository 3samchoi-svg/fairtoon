-- 32. 카카오페이지 작가명 4건 — 승인 목록 밖에서 새로 발견된 것
--
-- 08-31에 49편을 전수 대조하다가 나왔다. 7건 검수표에는 없던 항목이라 31번과 분리했다.
-- **31번을 먼저 돌리고, 아래 내용에 동의하시면 이 파일을 돌린다.** 순서 의존성은 없다.
--
-- 판정 근거: 카카오페이지는 역할을 밝히지 않고 이름만 나열한다(예: 리노,윤슬).
--            같은 플랫폼에서 웹소설판을 찾으면 원작자를 특정할 수 있다.
--            웹소설 크레딧에 있는 이름 = 원작, 웹툰에만 있는 이름 = 작화·각색.


-- ── 1. 황제의 외동딸 — 작화가 이름이 틀렸다 ──────────────────────────
-- 웹툰 '리노,윤슬' / 웹소설 '윤슬' → 원작 윤슬, 작화 리노
-- DB에는 작화가 '로담'으로 들어 있다. 근거를 찾지 못했다.

update public.works set author = '윤슬', art_author = '리노' where id = 49;


-- ── 2. 사내맞선 — 작화가 두 명이 빠져 있다 ───────────────────────────
-- 웹툰 '들깨,NARAK,해화' / 웹소설 '해화' → 원작 해화, 나머지 둘은 웹툰 제작진
-- 셋 중 누가 작화이고 누가 각색인지는 공개되지 않아 나누지 않는다.

update public.works set author = '들깨·NARAK·해화' where id = 46;


-- ── 3. 템빨 — 웹툰 제작진이 빠져 있다 ────────────────────────────────
-- 웹툰 '이동욱(REDICE STUDIO),Team Argo,박새날' / 웹소설 '박새날'

update public.works set author = '이동욱(REDICE STUDIO)·Team Argo·박새날' where id = 48;


-- ── 4. 나 혼자만 레벨업 — 각색자와 스튜디오 표기가 빠져 있다 ─────────
-- 웹툰 '현군,장성락(REDICE STUDIO),추공' / 웹소설 '추공'
-- DB의 '추공·장성락'도 틀린 것은 아니지만 플랫폼 표기와 다르다.

update public.works set author = '현군·장성락(REDICE STUDIO)·추공' where id = 45;


-- ── 확인 ──────────────────────────────────────────────────────────────

select id, title_ko, author, art_author
from public.works
where id in (45,46,48,49)
order by id;

-- 기대값
--   45 나 혼자만 레벨업  현군·장성락(REDICE STUDIO)·추공   (null)
--   46 사내맞선          들깨·NARAK·해화                    (null)
--   48 템빨              이동욱(REDICE STUDIO)·Team Argo·박새날  (null)
--   49 황제의 외동딸     윤슬                               리노


-- 이 4건을 31번과 나눈 이유
--
-- 승인받은 것은 검수표의 7건입니다. 위 4건은 같은 규칙(플랫폼 표기를 따른다)을 49편 전체에 적용하다가 새로 나온 것이라 승인 범위 밖입니다. 규칙이 같으니 함께 넣어도 될 일이지만, 승인 없이 데이터를 바꾼 것으로 남기지 않으려고 분리했습니다.
--
-- 동의하지 않으시는 항목이 있으면 그 블록만 지우고 Run 하시면 됩니다.
--
--
-- 판단이 갈릴 수 있는 지점 두 가지
--
-- ① 스튜디오 표기 — 2026-09-02 대표님 승인. 장성락(REDICE STUDIO)·이동욱(REDICE STUDIO)를 플랫폼 표기 그대로 쓴다.
--
-- ② 3인 크레딧을 나누지 않은 것 — 사내맞선·템빨·나혼렙은 이름이 셋인데, 카카오페이지가 역할을 밝히지 않아 누가 작화이고 누가 각색인지 확인할 방법이 없습니다. 웹소설 대조로 원작자만 특정됩니다. 그래서 author 한 칸에 셋을 그대로 두었습니다. 추측으로 나누면 창작자 크레딧을 우리가 임의로 바꾸는 셈이 되어 하지 않았습니다.
--
-- 2인 크레딧인 황제의 외동딸·김비서(31번)는 웹소설 대조로 원작과 작화가 명확히 갈려 나눴습니다.
